```sql
  IF v_trees <> c_expected_trees THEN
    RAISE EXCEPTION 'VERIFICACION FALLIDA: arboles esperados=%, encontrados=%',
      c_expected_trees, v_trees;
  END IF;
  IF v_null_subids <> 0 THEN
    RAISE EXCEPTION 'VERIFICACION FALLIDA: % arboles con sub_id NULL o vacio', v_null_subids;
  END IF;

  END IF;
  IF v_sincronizada <> 0 THEN
    RAISE EXCEPTION 'VERIFICACION FALLIDA: % grupos quedaron en estado sincronizada', v_sincronizada;
  END IF;
  IF v_dup_subid <> 0 THEN
    RAISE EXCEPTION 'VERIFICACION FALLIDA: % SubIDs duplicados dentro de su plantacion', v_dup_subid;
  END IF;
  IF v_orphan_groups <> 0 THEN
    RAISE EXCEPTION 'VERIFICACION FALLIDA: % grupos sin parcela_id (huerfanos)', v_orphan_groups;
  END IF;

  RAISE NOTICE 'Verificacion exitosa: % plantaciones, % parcelas, % grupos, % arboles, 0 huerfanos, 0 duplicados',
    v_plantations, v_parcelas, v_groups, v_trees;
END $$;

COMMIT;
```

```bash
  echo "ERROR: DATABASE_URL no definido. Exportarlo desde Supabase dashboard."

run_check() {
  local label="$1"
  local sql="$2"
  local expected="$3"
  local actual
  actual=$(psql "$DATABASE_URL" -t -A -c "$sql" 2>&1 | tr -d '[:space:]')
  if [[ "$actual" == "$expected" ]]; then
    echo "OK    | $label (got: $actual)"
  else
    echo "FAIL  | $label (expected: $expected, got: $actual)"
    exit 1
  fi
}

run_check "Trees count = $EXPECTED_TREES" \
```
